<script>
(function () {
  // Update this URL if the file is stored elsewhere. URLs are case-sensitive.
  var fileUrl = 'https://maggiexheuw.github.io/PPTX/adward.pptx';
  // NOTE: This is a UI-level gate, NOT secure protection of a publicly hosted file.
  var accessCode = '31415926';
  var openBtn = document.getElementById('award-open');
  var closeBtn = document.getElementById('award-close');
  var modal = document.getElementById('award-modal');
  var form = document.getElementById('award-form');
  var input = document.getElementById('award-password');
  var error = document.getElementById('award-error');
  if (!openBtn || !modal || !form) return;

  function closeModal() {
    modal.hidden = true;
    input.value = '';
    error.textContent = '';
    openBtn.focus();
  }
  openBtn.addEventListener('click', function () {
    modal.hidden = false;
    error.textContent = '';
    input.focus();
  });
  closeBtn.addEventListener('click', closeModal);
  modal.addEventListener('click', function (e) { if (e.target === modal) closeModal(); });
  document.addEventListener('keydown', function (e) {
    if (modal.hidden) return;
    if (e.key === 'Escape') closeModal();
    if (e.key === 'Tab') {
      var stops = [closeBtn, input, form.querySelector('button[type="submit"]')];
      var i = stops.indexOf(document.activeElement);
      if (e.shiftKey && i === 0) { e.preventDefault(); stops[2].focus(); }
      else if (!e.shiftKey && i === 2) { e.preventDefault(); stops[0].focus(); }
    }
  });
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    if (input.value === accessCode) {
      window.open(fileUrl, '_blank', 'noopener,noreferrer');
      closeModal();
    } else {
      error.textContent = 'Incorrect password. Please try again.';
      input.select();
    }
  });
})();
</script>
