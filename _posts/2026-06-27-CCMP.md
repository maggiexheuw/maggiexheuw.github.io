(function () {
  const page = document.getElementById('ccmp-top');
  if (!page) return;

  // Theme toggle
  const themeToggle = document.getElementById('ccmp-theme-toggle');
  const savedTheme = localStorage.getItem('ccmp-theme');
  if (savedTheme === 'dark') {
    page.setAttribute('data-ccmp-theme', 'dark');
  }

  themeToggle?.addEventListener('click', () => {
    const isDark = page.getAttribute('data-ccmp-theme') === 'dark';
    if (isDark) {
      page.removeAttribute('data-ccmp-theme');
      localStorage.setItem('ccmp-theme', 'light');
    } else {
      page.setAttribute('data-ccmp-theme', 'dark');
      localStorage.setItem('ccmp-theme', 'dark');
    }
  });

  // Search & Filter
  const searchInput = document.getElementById('ccmp-search-input');
  const filterButtons = document.querySelectorAll('.ccmp-filter');
  const tagButtons = document.querySelectorAll('.ccmp-tag');
  const cards = document.querySelectorAll('.ccmp-card');
  const resultCount = document.getElementById('ccmp-result-count');
  const emptyState = document.getElementById('ccmp-empty');

  let currentFilter = 'all';
  let currentQuery = '';

  function updateReports() {
    let visibleCount = 0;
    const q = currentQuery.trim().toLowerCase();

    cards.forEach(card => {
      const tags = (card.getAttribute('data-tags') || '').toLowerCase();
      const text = card.textContent.toLowerCase();

      const matchesFilter = currentFilter === 'all' || tags.includes(currentFilter.toLowerCase());
      const matchesQuery = !q || text.includes(q);

      if (matchesFilter && matchesQuery) {
        card.classList.remove('is-hidden');
        visibleCount++;
      } else {
        card.classList.add('is-hidden');
      }
    });

    if (resultCount) {
      resultCount.textContent = `Showing ${visibleCount} report${visibleCount === 1 ? '' : 's'}`;
    }
    if (emptyState) {
      emptyState.hidden = visibleCount > 0;
    }
  }

  searchInput?.addEventListener('input', (e) => {
    currentQuery = e.target.value;
    updateReports();
  });

  filterButtons.forEach(btn => {
    btn.addEventListener('click', () => {
      filterButtons.forEach(b => b.classList.remove('is-active'));
      btn.classList.add('is-active');
      currentFilter = btn.getAttribute('data-filter') || 'all';
      updateReports();
    });
  });

  tagButtons.forEach(btn => {
    btn.addEventListener('click', (e) => {
      e.stopPropagation();
      const tagText = btn.getAttribute('data-filter');
      if (!tagText) return;

      filterButtons.forEach(b => {
        if (b.getAttribute('data-filter') === tagText) {
          b.click();
        }
      });
      window.scrollTo({ top: page.querySelector('.ccmp-tools').offsetTop - 30, behavior: 'smooth' });
    });
  });

  // PDF Preview Modal
  const modal = document.getElementById('ccmp-modal');
  const modalClose = document.getElementById('ccmp-modal-close');
  const modalIframe = document.getElementById('ccmp-modal-iframe');
  const modalTitle = document.getElementById('ccmp-modal-title');
  const previewButtons = document.querySelectorAll('.ccmp-preview');

  previewButtons.forEach(btn => {
    btn.addEventListener('click', () => {
      const pdfUrl = btn.getAttribute('data-pdf');
      const title = btn.getAttribute('data-title') || 'PDF Preview';
      if (!pdfUrl || !modal || !modalIframe) return;

      modalTitle.textContent = title;
      modalIframe.src = pdfUrl;
      modal.hidden = false;
      document.body.style.overflow = 'hidden';
    });
  });

  function closeModal() {
    if (!modal || !modalIframe) return;
    modal.hidden = true;
    modalIframe.src = 'about:blank';
    document.body.style.overflow = '';
  }

  modalClose?.addEventListener('click', closeModal);
  modal?.addEventListener('click', (e) => {
    if (e.target === modal) closeModal();
  });
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') closeModal();
  });
}());
</script>
